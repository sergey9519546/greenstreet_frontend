### The Fix: Nelson-Siegel-Svensson Rate Surface + Hull-White Short-Rate Sim
# Nelson-Siegel model:
y(tau) = b0 + b1 * [(1 - exp(-tau/lambda)) / (tau/lambda)]
              + b2 * [(1 - exp(-tau/lambda)) / (tau/lambda) - exp(-tau/lambda)]

# Svensson extension adds: + b3 * [(1 - exp(-tau/lambda2))/(tau/lambda2) - exp(-tau/lambda2)]

# Daily fit workflow:
# 1. Pull SOFR swap quotes at 1M, 3M, 6M, 1Y, 2Y, 3Y, 5Y, 7Y, 10Y from FRED
# 2. Fit (b0, b1, b2, b3, lambda1, lambda2) by nonlinear least squares
# 3. Evaluate forward rate at ARM reset maturity: f(tau) = -d/dtau [tau * y(tau)]
# 4. Add margin (e.g., 2.50%) to get projected reset rate
# 5. For Monte Carlo: fit Hull-White (one-factor) short-rate model for rate path uncertainty
from scipy.optimize import minimize
import numpy as np

def nelson_siegel(tau, b0, b1, b2, lam):
    factor1 = (1 - np.exp(-tau/lam)) / (tau/lam)
    factor2 = factor1 - np.exp(-tau/lam)
    return b0 + b1 * factor1 + b2 * factor2

def fit_ns(maturities, yields):

## 18. Sequential Drawdown + MCID + Counterfactual Generator
### 18.1 Sequential Drawdown Array (path-dependent)
def sequential_drawdown(monthly_rent, monthly_opex, monthly_ds,
                        capex_events, starting_reserves):
    balance = starting_reserves
    ruin_month = None
    min_balance = balance
    for t in range(36):
        ncf = monthly_rent[t] - monthly_opex[t] - monthly_ds[t]
        if t in capex_events: ncf -= capex_events[t]
        balance += ncf
        if balance < min_balance: min_balance = balance
        if balance < 0 and ruin_month is None: ruin_month = t + 1
    return {'liquidity_failure': ruin_month is not None,
            'ruin_month': ruin_month, 'min_balance': min_balance}
### 18.2 MCID — Max Cumulative Intra-Year Deficit (STR-specific)
def compute_mcid(monthly_net_cashflow):
    """Max cumulative deficit in any rolling 12-mo window"""
    cumulative = 0; min_cumulative = 0
    for cash in monthly_net_cashflow:
        cumulative += cash
        if cumulative < min_cumulative: min_cumulative = cumulative
    return abs(min_cumulative)

### 18.3 Counterfactual Generator (Binary Search for Minimum Change)
def counterfactual_search(target_state, current_state, all_variables,
                         constraints):
    """Find smallest change to any single input that flips verdict."""
    candidates = []
    for var in all_variables:
        # Binary search delta for this variable
        lo, hi = constraints[var]['min'], constraints[var]['max']
        while hi - lo > 0.01:  # precision
            mid = (lo + hi) / 2
            test_state = current_state.copy()
            test_state[var] = mid
            if simulate(test_state) == target_state:
                hi = mid
            else:
                lo = mid
        candidates.append({'variable': var, 'min_change': hi})
    return sorted(candidates, key=lambda c: c['min_change'])

## 31. Conformal Prediction Vault (Decaying Confidence)
### 31.1 The Framework (Beyond the Rulebook #2)
from sklearn.ensemble import GradientBoostingRegressor
import numpy as np

def conformal_prediction_interval(new_x, calibration_X, calibration_y,
                                base_model, alpha=0.10):
    # Step 1: Compute nonconformity scores on calibration set
    cal_pred = base_model.predict(calibration_X)
    scores = np.abs(calibration_y - cal_pred)

    # Step 2: Apply exponential decay (e^-lambda * t)
    # Tier 1 (county tax): lambda=low, decays 365d
    # Tier 3 (borrower-stated rent): lambda=high, decays 30d
    decay_factors = np.exp(-lambda * data_age_days)
    weighted_scores = scores * decay_factors

    # Step 3: Quantile for (1-alpha) coverage
    q_hat = np.quantile(weighted_scores, np.ceil((len(scores)+1)*(1-alpha))/len(scores))

    # Step 4: Output interval
    point_pred = base_model.predict([new_x])[0]
    return (point_pred - q_hat, point_pred + q_hat)
